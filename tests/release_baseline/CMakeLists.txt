add_executable(umicom-release-evidence-tests "${CMAKE_CURRENT_LIST_DIR}/test_evidence.c")
target_link_libraries(umicom-release-evidence-tests PRIVATE Umicom::distribution)
set_target_properties(umicom-release-evidence-tests PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(COMMAND umicom_apply_warnings)
    umicom_apply_warnings(umicom-release-evidence-tests)
endif()
if(COMMAND umicom_apply_sanitizers)
    umicom_apply_sanitizers(umicom-release-evidence-tests)
endif()
set(_release_test_names)
foreach(_case IN ITEMS missing full subset candidate source profile configuration kind failed skipped not_run false_pass zero digest reference unknown)
    add_test(NAME framework.release_baseline.evaluate.${_case} COMMAND umicom-release-evidence-tests evaluate ${_case})
    list(APPEND _release_test_names framework.release_baseline.evaluate.${_case})
endforeach()
foreach(_case IN ITEMS null capacity over_capacity missing_category duplicate_contract duplicate_evidence nul control non_ascii large_input crlf no_final_lf owned live_output long_line bad_header negative overflow sum_overflow sum_mismatch bad_hex path_parent path_absolute unknown_kind unknown_outcome)
    add_test(NAME framework.release_baseline.parser.${_case} COMMAND umicom-release-evidence-tests parser ${_case})
    list(APPEND _release_test_names framework.release_baseline.parser.${_case})
endforeach()
add_test(NAME framework.release_baseline.mutations COMMAND umicom-release-evidence-tests mutate all)
add_test(NAME framework.release_baseline.lesson COMMAND umicom-release-baseline-example)
add_test(NAME framework.release_baseline.cli_help COMMAND umicom-release-baseline --help)
add_test(NAME framework.release_baseline.contract_load COMMAND umicom-release-baseline show "${CMAKE_CURRENT_LIST_DIR}/../../docs/releases/core/CORE_REQUIREMENTS.tsv")
add_test(NAME framework.release_baseline.pending_refused COMMAND ${CMAKE_COMMAND} "-DTOOL=$<TARGET_FILE:umicom-release-baseline>" "-DCORE=${CMAKE_CURRENT_LIST_DIR}/../../docs/releases/core" -P "${CMAKE_CURRENT_LIST_DIR}/check_pending.cmake")
list(APPEND _release_test_names framework.release_baseline.mutations framework.release_baseline.lesson framework.release_baseline.cli_help framework.release_baseline.contract_load framework.release_baseline.pending_refused)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_executable(umicom-release-evidence-allocation "${CMAKE_CURRENT_LIST_DIR}/test_allocation.c")
    target_link_libraries(umicom-release-evidence-allocation PRIVATE Umicom::distribution)
    target_link_options(umicom-release-evidence-allocation PRIVATE -Wl,--wrap=calloc)

    if(COMMAND umicom_apply_warnings)
        umicom_apply_warnings(umicom-release-evidence-allocation)
    endif()
    if(COMMAND umicom_apply_sanitizers)
        umicom_apply_sanitizers(umicom-release-evidence-allocation)
    endif()
    add_test(NAME framework.release_baseline.allocation COMMAND umicom-release-evidence-allocation)
    list(APPEND _release_test_names framework.release_baseline.allocation)
endif()

"-DDEST=${CMAKE_CURRENT_BINARY_DIR}/release-composition-${_mode}" "-DDEFER=${_mode}" "-DCOMPILER=${CMAKE_C_COMPILER}" -P "${CMAKE_CURRENT_LIST_DIR}/check_composition.cmake")
        list(APPEND _release_test_names framework.release_baseline.composition.${_mode})
    endforeach()
