cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(UmicomDenoisingChecks LANGUAGES C)
    include(CTest)
    get_filename_component(_denoising_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    add_library(denoising_subject STATIC "${_denoising_root}/src/ai/training/denoising.c")
    target_include_directories(denoising_subject PUBLIC "${_denoising_root}/include")
    target_compile_features(denoising_subject PUBLIC c_std_23)
    if(NOT WIN32)
        target_link_libraries(denoising_subject PUBLIC m)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(denoising_subject PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
    set(_denoising_library denoising_subject)
else()
    set(_denoising_library Umicom::ai)
endif()
add_executable(umicom-denoising-test test_denoising.c)
target_link_libraries(umicom-denoising-test PRIVATE ${_denoising_library})
target_compile_features(umicom-denoising-test PRIVATE c_std_23)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(umicom-denoising-test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(COMMAND umicom_register_validation_target)
    umicom_register_validation_target(umicom-denoising-test)
endif()
foreach(_case reference roundtrip endpoints nonfinite overflow alpha capacity arguments)
    add_test(NAME framework.ai.denoising.${_case} COMMAND umicom-denoising-test ${_case})
    set_tests_properties(framework.ai.denoising.${_case} PROPERTIES TIMEOUT 20 LABELS "framework;ai;training;denoising")
    if(COMMAND umicom_test_impact_owners)
        umicom_test_impact_owners(framework.ai.denoising.${_case} TARGETS umicom-denoising-test)
    endif()
endforeach()
