umicom_creative_configure_target(umicom-creative-audio-test)
include("${CMAKE_CURRENT_LIST_DIR}/cases.cmake")
foreach(case IN LISTS UMICOM_CREATIVE_AUDIO_CASES)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-tests/${case}")
    add_test(NAME "framework.creative_audio.${case}" COMMAND umicom-creative-audio-test "${case}"
        "${CMAKE_CURRENT_BINARY_DIR}/audio-tests/${case}")
    set_tests_properties("framework.creative_audio.${case}" PROPERTIES TIMEOUT 30
        LABELS "framework;creative-audio;regression")
endforeach()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_executable(umicom-creative-audio-allocation-test "${CMAKE_CURRENT_LIST_DIR}/test_alloc.c")
    target_link_libraries(umicom-creative-audio-allocation-test PRIVATE Umicom::creative_workspace)
    target_link_options(umicom-creative-audio-allocation-test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc)
    umicom_creative_configure_target(umicom-creative-audio-allocation-test)
    add_test(NAME framework.creative_audio.allocation_failures COMMAND umicom-creative-audio-allocation-test)
    set_tests_properties(framework.creative_audio.allocation_failures PROPERTIES TIMEOUT 30 LABELS "framework;creative-audio;allocation")
    foreach(case symlink fifo directory)
        add_test(NAME "framework.creative_audio.file_${case}" COMMAND umicom-creative-audio-test "file_${case}" "${CMAKE_CURRENT_BINARY_DIR}/audio-tests")
        set_tests_properties("framework.creative_audio.file_${case}" PROPERTIES TIMEOUT 10 LABELS "framework;creative-audio;file")
    endforeach()
    if(COMMAND umicom_register_validation_target)
        umicom_register_validation_target(umicom-creative-audio-allocation-test)
    endif()
endif()
