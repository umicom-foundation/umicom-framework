# Real CTest catalogue; discovery must never execute its marker-writing bodies.
cmake_minimum_required(VERSION 3.21)
project(UmicomJsonDiscoveryFixture NONE)
enable_testing()
if(NOT DEFINED CASE_COUNT)
    set(CASE_COUNT 10000)
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/working notes")
if(CASE_COUNT GREATER 0)
    math(EXPR last "${CASE_COUNT} - 1")
    foreach(index RANGE 0 ${last})
        if(index EQUAL 1)
            string(REPEAT "x" 200 suffix)
            set(name "notes.${suffix}")
        elseif(index EQUAL 2)
            set(name "notes.literal (Disabled)")
        elseif(index EQUAL 3)
            set(name "notes.é.日本")
        else()
            set(name "notes.case${index}")
        endif()
        add_test(NAME "${name}" COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_BINARY_DIR}/unexpected-test-execution.marker")
        set_tests_properties("${name}" PROPERTIES LABELS "learning;notes" TIMEOUT 1.25 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/working notes")
        math(EXPR remainder "${index} % 100")
        if(remainder EQUAL 0)
            set_tests_properties("${name}" PROPERTIES DISABLED TRUE)
        endif()
    endforeach()
endif()
