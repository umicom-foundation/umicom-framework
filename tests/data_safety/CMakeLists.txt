# Umicom Framework | Sammy Hegab, Umicom Foundation | MIT
# Tests compile canonical data_server.c through a laboratory translation unit
# solely for SQLite failure injection. The public example links Umicom::data.
find_package(Threads REQUIRED)
add_executable(umicom-data-safety-test test_data_safety.c)
target_include_directories(umicom-data-safety-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
target_link_libraries(umicom-data-safety-test PRIVATE Threads::Threads)
option(UMICOM_DATA_SAFETY_SQLITE "Enable SQLite failure-injection cases when available" ON)
if(UMICOM_DATA_SAFETY_SQLITE)
    find_package(SQLite3 QUIET)
    if(SQLite3_FOUND)
        target_compile_definitions(umicom-data-safety-test PRIVATE UMICOM_HAS_SQLITE=1)
        target_link_libraries(umicom-data-safety-test PRIVATE SQLite::SQLite3)
    endif()
endif()
set_target_properties(umicom-data-safety-test PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(COMMAND umicom_apply_warnings)
    umicom_apply_warnings(umicom-data-safety-test)
endif()
if(COMMAND umicom_apply_sanitizers)
    umicom_apply_sanitizers(umicom-data-safety-test)
endif()
set(_cases roundtrip rollback commit nested state small_read missing invalid snapshot visit owner_retired repeated two_servers)
foreach(_number RANGE 0 9)
    list(APPEND _cases "compete_${_number}" "reenter_${_number}")
endforeach()
foreach(_backend memory sqlite)
    foreach(_case IN LISTS _cases)
        add_test(NAME "framework.data_safety.${_backend}.${_case}" COMMAND umicom-data-safety-test "${_backend}" "${_case}")
        set_tests_properties("framework.data_safety.${_backend}.${_case}" PROPERTIES TIMEOUT 20 SKIP_RETURN_CODE 77)
    endforeach()
endforeach()
foreach(_case bind_get bind_delete bind_set rollback_denied commit_denied automatic_abort raw_transaction count_error diagnostic embedded_nul)
    add_test(NAME "framework.data_safety.sqlite.${_case}" COMMAND umicom-data-safety-test sqlite "${_case}")
    set_tests_properties("framework.data_safety.sqlite.${_case}" PROPERTIES TIMEOUT 20 SKIP_RETURN_CODE 77)
endforeach()
