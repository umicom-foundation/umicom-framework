# Include-safe definitions: never create a subdirectory from a deferred callback.
set(_study_tests "${CMAKE_CURRENT_LIST_DIR}")
add_executable(umicom-education-study-test "${_study_tests}/test_study.c")
target_include_directories(umicom-education-study-test PRIVATE "${_study_tests}/../../src/education_workspace")
target_link_libraries(umicom-education-study-test PRIVATE Umicom::education_workspace)
umicom_education_configure_target(umicom-education-study-test)
include("${_study_tests}/cases.cmake")
foreach(case IN LISTS UMICOM_EDUCATION_STUDY_CASES)
    add_test(NAME "framework.education_study.${case}" COMMAND umicom-education-study-test "${case}")
    set_tests_properties("framework.education_study.${case}" PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 LABELS "framework;education-study;regression")
endforeach()
add_test(NAME framework.education_study.lesson COMMAND umicom-study --self-test)
add_test(NAME framework.education_study.library_cli COMMAND umicom-study library PCM16)
set_tests_properties(framework.education_study.lesson framework.education_study.library_cli PROPERTIES TIMEOUT 30 LABELS "framework;education-study;example")
if(COMMAND umicom_register_validation_target)
    umicom_register_validation_target(umicom-education-study-test)
    umicom_register_validation_target(umicom-study)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_executable(umicom-education-study-alloc "${_study_tests}/test_alloc.c")
    target_link_libraries(umicom-education-study-alloc PRIVATE Umicom::education_workspace)
    target_link_options(umicom-education-study-alloc PRIVATE -Wl,--wrap=calloc)
    umicom_education_configure_target(umicom-education-study-alloc)
    add_test(NAME framework.education_study.allocation_failure COMMAND umicom-education-study-alloc)
    set_tests_properties(framework.education_study.allocation_failure PROPERTIES TIMEOUT 30 LABELS "framework;education-study;allocation")
    if(COMMAND umicom_register_validation_target)
        umicom_register_validation_target(umicom-education-study-alloc)
    endif()
endif()
if(TARGET umicom_education_workspace_gtk4)
    add_executable(umicom-education-study-gtk-test "${_study_tests}/test_gtk.c")
    target_link_libraries(umicom-education-study-gtk-test PRIVATE Umicom::education_workspace_gtk4)
    umicom_education_configure_target(umicom-education-study-gtk-test)
    foreach(case capture library invalidate retained_control unsaved_note)
        add_test(NAME "framework.education_study.gtk.${case}" COMMAND umicom-education-study-gtk-test "${case}")
        set_tests_properties("framework.education_study.gtk.${case}" PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77 LABELS "framework;education-study;gtk4")
    endforeach()
    if(COMMAND umicom_register_validation_target)
        umicom_register_validation_target(umicom-education-study-gtk-test)
    endif()
endif()
unset(_study_tests)
