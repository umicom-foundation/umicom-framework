function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(PROBE_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
endfunction()

# Validation itself only inspects already-registered targets; it creates no
# directories, substitutes no dependencies and opens no broker connections.
function(VerifyIntegration)
    set(_expected)
    if(MODE STREQUAL "partial")
        set(_expected ibkr_connection)
    elseif(NOT MODE STREQUAL "missing")
        set(_expected market_tape research_replay ibkr_connection)
    endif()
    foreach(_component IN ITEMS market_tape research_replay ibkr_connection)
        if(_component IN_LIST _expected)
            if(NOT TARGET Umicom::${_component})
                message(FATAL_ERROR "Missing composed target Umicom::${_component}")
            endif()
            get_target_property(_sources umicom_${_component} SOURCES)
            foreach(_source IN LISTS _sources)
                if(NOT IS_ABSOLUTE "${_source}" OR NOT EXISTS "${_source}")
                    message(FATAL_ERROR "Unrooted or missing component source: ${_source}")
                endif()
            endforeach()
        elseif(TARGET Umicom::${_component})
            message(FATAL_ERROR "Unexpected component without its canonical owners: ${_component}")
        endif()
    endforeach()
    get_property(_tests DIRECTORY PROPERTY TESTS)
    set(_sorted "${_tests}")
    list(LENGTH _sorted _length)
    list(REMOVE_DUPLICATES _sorted)
    list(LENGTH _sorted _unique)
    if(NOT _length EQUAL _unique)
        message(FATAL_ERROR "Duplicate test registration")
    endif()
    list(SORT _sorted)
    string(REPLACE ";" "\n" _text "${_sorted}")
    file(WRITE "${CMAKE_BINARY_DIR}/registered-tests.txt" "${_text}\n")
    file(WRITE "${CMAKE_BINARY_DIR}/integration-result.txt" "mode=${MODE}\ncomponents=${_expected}\ntests=${_length}\nsystem=${CMAKE_SYSTEM_NAME}\ncompiler=${CMAKE_C_COMPILER_ID}\n")
endfunction()
cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL VerifyIntegration)
