cmake_minimum_required(VERSION 3.24)
project(UmicomBankCompositionFixture LANGUAGES C)
include(CTest)
# This fixture qualifies CMake target composition only. Imported dependency
# names stand in for the surrounding application; no executable is built/run.
add_library(Umicom::finance INTERFACE IMPORTED)
add_library(Umicom::data INTERFACE IMPORTED)
function(umicom_apply_warnings target)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "Warning policy received a missing target")
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(NOT TARGET "${target}")
        message(FATAL_ERROR "Sanitizer policy received a missing target")
    endif()
endfunction()
# The owner must compose every extension after creating its target and helpers.
include("${ROOT}/cmake/UmicomBankOperations.cmake")
include("${ROOT}/cmake/UmicomBankOperations.cmake")
if(NOT TARGET Umicom::bank_operations OR NOT COMMAND umicom_bank_review_target)
    message(FATAL_ERROR "The optional bank owner did not compose its review boundary")
endif()
get_target_property(_sources umicom_bank_operations SOURCES)
foreach(_leaf activity_query.c activity.c activity_text.c activity_csv.c reservations.c reservations_text.c reservations_csv.c
        audit_query.c audit_report.c audit_text.c audit_csv.c)
    set(_count 0)
    foreach(_source IN LISTS _sources)
        cmake_path(GET _source FILENAME _name)
        if(_name STREQUAL _leaf)
            math(EXPR _count "${_count} + 1")
        endif()
    endforeach()
    if(NOT _count EQUAL 1)
        message(FATAL_ERROR "Expected one canonical bank implementation for ${_leaf}, found ${_count}")
    endif()
endforeach()
get_property(_tests DIRECTORY PROPERTY TESTS)
foreach(_test framework.bank_activity.capture.all framework.bank_reservations.capture.all framework.bank_audit.capture.all)
    if(BUILD_TESTING AND NOT _test IN_LIST _tests)
        message(FATAL_ERROR "Missing composed banking regression: ${_test}")
    elseif(NOT BUILD_TESTING AND _test IN_LIST _tests)
        message(FATAL_ERROR "Disabled tests leaked into bank composition: ${_test}")
    endif()
endforeach()
