cmake_minimum_required(VERSION 3.24)
project(UmicomQueueComposition LANGUAGES C)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(UMICOM_QUEUE_SHUTDOWN_COMPOSITION_CHILD ON)
if(NOT DEFINED FRAMEWORK OR NOT DEFINED MODE)
    message(FATAL_ERROR "Pass canonical FRAMEWORK and composition MODE")
endif()
if(MODE STREQUAL "minimal")
    add_library(umicom_platform INTERFACE)
    add_library(Umicom::platform ALIAS umicom_platform)
    include("${FRAMEWORK}/cmake/UmicomQueueShutdown.cmake")
    if(TARGET umicom-queue-shutdown)
        message(FATAL_ERROR "Minimal host acquired an unrequested task runtime")
    endif()
    return()
endif()
add_library(umicom_platform STATIC "${FRAMEWORK}/src/platform/threading.c")
add_library(Umicom::platform ALIAS umicom_platform)
set_target_properties(umicom_platform PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(umicom_platform PUBLIC "${FRAMEWORK}/include")
target_link_libraries(umicom_platform PUBLIC Threads::Threads)
if(MODE STREQUAL "late_sources")
    include("${FRAMEWORK}/cmake/UmicomQueueShutdown.cmake")
    if(TARGET umicom-queue-shutdown)
        message(FATAL_ERROR "Incomplete platform acquired queue consumers")
    endif()
endif()
target_sources(umicom_platform PRIVATE
    "${FRAMEWORK}/src/platform/task.c" "${FRAMEWORK}/src/platform/task_queue.c"
    "${FRAMEWORK}/src/platform/clock.c" "${FRAMEWORK}/src/platform/cancellation.c")
function(Complete)
    include("${FRAMEWORK}/cmake/UmicomQueueShutdown.cmake")
    include("${FRAMEWORK}/cmake/UmicomQueueShutdown.cmake")
endfunction()
if(MODE STREQUAL "immediate")
    Complete()
else()
    cmake_language(DEFER CALL Complete)
endif()
