# Native primary regressions: no interpreter or command-string execution.
add_executable(umicom-native-launcher-test test_main.c test_sha.c test_manifest.c test_formats.c)
target_link_libraries(umicom-native-launcher-test PRIVATE Umicom::native_launcher)
target_include_directories(umicom-native-launcher-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src/native_launcher")
set_target_properties(umicom-native-launcher-test PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
set(_case_files cases_common.inc)
set(_child_argument "")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(umicom-native-launcher-test PRIVATE test_stage_linux.c)
    add_executable(umicom-native-launcher-inert-child inert_child.c)
    set_target_properties(umicom-native-launcher-inert-child PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    list(APPEND _case_files cases_linux.inc)
    set(_child_argument "$<TARGET_FILE:umicom-native-launcher-inert-child>")
    add_dependencies(umicom-native-launcher-test umicom-native-launcher-inert-child)
    if(COMMAND umicom_apply_warnings)
        umicom_apply_warnings(umicom-native-launcher-inert-child)
    endif()
endif()
if(COMMAND umicom_apply_warnings)
    umicom_apply_warnings(umicom-native-launcher-test)
endif()
if(COMMAND umicom_apply_sanitizers)
    umicom_apply_sanitizers(umicom-native-launcher-test)
endif()
foreach(_file IN LISTS _case_files)
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/${_file}" _lines REGEX "^CASE")
    foreach(_line IN LISTS _lines)
        string(REGEX REPLACE "^CASE\\(\"([^\"]+)\",.*" "\\1" _case "${_line}")
        add_test(NAME "framework.native_launcher.${_case}"
            COMMAND umicom-native-launcher-test "${_case}" "${_child_argument}")
        set_tests_properties("framework.native_launcher.${_case}" PROPERTIES
            TIMEOUT 30 LABELS "framework;native-launcher;c23;regression")
    endforeach()
endforeach()
if(TARGET umicom-session-stage)
    add_test(NAME framework.native_launcher.tool.help COMMAND umicom-session-stage --help)
    add_test(NAME framework.native_launcher.tool.self_test COMMAND umicom-session-stage --self-test)
    set_tests_properties(framework.native_launcher.tool.help framework.native_launcher.tool.self_test
        PROPERTIES TIMEOUT 30 LABELS "framework;native-launcher;c23;regression")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/digest-fixture.txt" "abc")
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR WIN32)
        add_test(NAME framework.native_launcher.tool.digest COMMAND umicom-session-stage digest
            --file "${CMAKE_CURRENT_BINARY_DIR}/digest-fixture.txt")
        set_tests_properties(framework.native_launcher.tool.digest PROPERTIES TIMEOUT 30
            LABELS "framework;native-launcher;c23;regression"
            PASS_REGULAR_EXPRESSION "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad")
    endif()
endif()
if(COMMAND umicom_register_validation_target)
    umicom_register_validation_target(umicom-native-launcher-test)
    if(TARGET umicom-session-stage)
        umicom_register_validation_target(umicom-session-stage)
    endif()
endif()
