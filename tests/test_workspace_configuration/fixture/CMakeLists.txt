cmake_minimum_required(VERSION 3.28)
project(UmicomTestWorkspaceFixture LANGUAGES NONE)
enable_testing()
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
# Clear only this miniature fixture's own cache setting between repeated runs.
unset(UMICOM_TEST_WORK_ROOT CACHE)
if(CASE STREQUAL "custom")
    set(UMICOM_TEST_WORK_ROOT "${CMAKE_BINARY_DIR}/accessible work" CACHE PATH "" FORCE)
elseif(CASE STREQUAL "relative")
    set(UMICOM_TEST_WORK_ROOT "relative-work" CACHE PATH "" FORCE)
elseif(CASE STREQUAL "invalid-empty")
    set(UMICOM_TEST_WORK_ROOT "" CACHE PATH "" FORCE)
elseif(CASE STREQUAL "invalid-list")
    set(UMICOM_TEST_WORK_ROOT "bad;root" CACHE PATH "" FORCE)
elseif(CASE STREQUAL "invalid-generator")
    set(UMICOM_TEST_WORK_ROOT "$<CONFIG>/work" CACHE PATH "" FORCE)
endif()
include("${_framework}/cmake/UmicomTestWorkspace.cmake")
add_test(NAME fixture.case COMMAND "${CMAKE_COMMAND}" -E true)
set_property(TEST fixture.case PROPERTY ENVIRONMENT "KEEP_SETTING=present")
set_property(TEST fixture.case PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:retained-path")
if(CASE STREQUAL "explicit-environment")
    set_property(TEST fixture.case APPEND PROPERTY ENVIRONMENT "TEMP=explicit-temp")
elseif(CASE STREQUAL "explicit-modification")
    set_property(TEST fixture.case APPEND PROPERTY ENVIRONMENT_MODIFICATION "TMPDIR=set:explicit-temp")
endif()
umicom_configure_test_workspace(fixture.case "${CMAKE_CURRENT_SOURCE_DIR}")
get_property(_environment TEST fixture.case PROPERTY ENVIRONMENT)
get_property(_modifications TEST fixture.case PROPERTY ENVIRONMENT_MODIFICATION)
if(NOT "KEEP_SETTING=present" IN_LIST _environment OR
   NOT "PATH=path_list_prepend:retained-path" IN_LIST _modifications)
    message(FATAL_ERROR "Unrelated fixture environment was changed")
endif()
if(CASE MATCHES "^explicit-")
    foreach(_entry IN LISTS _environment)
        if(_entry MATCHES "^(TMPDIR|TMP|UMICOM_TEST_WORK_DIRECTORY)=")
            message(FATAL_ERROR "The explicit temp policy was overwritten")
        endif()
    endforeach()
    return()
endif()
set(_work)
foreach(_entry IN LISTS _environment)
    if(_entry MATCHES "^UMICOM_TEST_WORK_DIRECTORY=(.*)$")
        set(_work "${CMAKE_MATCH_1}")
    endif()
endforeach()
if(NOT IS_DIRECTORY "${_work}" OR NOT IS_ABSOLUTE "${_work}")
    message(FATAL_ERROR "No absolute retained fixture directory was created")
endif()
foreach(_name TMPDIR TMP TEMP)
    if(NOT "${_name}=${_work}" IN_LIST _environment)
        message(FATAL_ERROR "Temporary directory aliases do not match")
    endif()
endforeach()
if(CASE STREQUAL "default" AND NOT _work MATCHES "/test-work/")
    message(FATAL_ERROR "Default fixtures escaped the build-local test-work root")
elseif(CASE STREQUAL "custom" AND NOT _work MATCHES "/accessible work/")
    message(FATAL_ERROR "Custom accessible root was ignored")
elseif(CASE STREQUAL "relative" AND NOT _work MATCHES "/relative-work/")
    message(FATAL_ERROR "Relative root was not resolved under this build")
endif()
if(CASE STREQUAL "idempotent")
    umicom_configure_test_workspace(fixture.case "${CMAKE_CURRENT_SOURCE_DIR}")
    get_property(_again TEST fixture.case PROPERTY ENVIRONMENT)
    if(NOT _again STREQUAL _environment)
        message(FATAL_ERROR "Repeated registration duplicated environment entries")
    endif()
elseif(CASE STREQUAL "collision")
    add_test(NAME fixture-case COMMAND "${CMAKE_COMMAND}" -E true)
    umicom_configure_test_workspace(fixture-case "${CMAKE_CURRENT_SOURCE_DIR}")
    get_property(_other TEST fixture-case PROPERTY ENVIRONMENT)
    if("UMICOM_TEST_WORK_DIRECTORY=${_work}" IN_LIST _other)
        message(FATAL_ERROR "Different names shared a fixture directory")
    endif()
elseif(CASE STREQUAL "separate-directory")
    add_subdirectory(child)
    umicom_configure_test_workspaces("${CMAKE_BINARY_DIR}")
    get_property(_other TEST fixture.case DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/child" PROPERTY ENVIRONMENT)
    if("UMICOM_TEST_WORK_DIRECTORY=${_work}" IN_LIST _other)
        message(FATAL_ERROR "Tests in different directories shared a fixture directory")
    endif()
endif()
