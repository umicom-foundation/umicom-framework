# Umicom Foundation | MIT. Model an ordinary or deferred application composition.
cmake_minimum_required(VERSION 3.24)
project(UmicomThreadComposition LANGUAGES C)
include(CTest)
find_package(Threads REQUIRED)
set(UMICOM_THREAD_SAFETY_INTEGRATION_CHILD ON)
set(UMICOM_THREAD_SAFETY_STRICT ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MODE STREQUAL "minimal")
    add_library(umicom_platform INTERFACE)
else()
    add_library(umicom_platform STATIC "${FRAMEWORK}/src/platform/threading.c")
    target_include_directories(umicom_platform PUBLIC "${FRAMEWORK}/include")
    target_link_libraries(umicom_platform PUBLIC Threads::Threads)
    set_target_properties(umicom_platform PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
add_library(Umicom::platform ALIAS umicom_platform)
function(Complete)
    include("${FRAMEWORK}/cmake/UmicomThreadSafety.cmake")
    include("${FRAMEWORK}/cmake/UmicomThreadSafety.cmake")
    if(MODE STREQUAL "minimal")
        if(TARGET umicom-thread-lifecycle OR TARGET umicom-test-thread-safety)
            message(FATAL_ERROR "A minimal platform acquired a substitute thread runtime")
        endif()
    elseif(NOT TARGET umicom-test-thread-safety)
        message(FATAL_ERROR "Canonical thread tests were not registered")
    endif()
endfunction()
if(MODE STREQUAL "deferred")
    cmake_language(DEFER CALL Complete)
else()
    Complete()
endif()
