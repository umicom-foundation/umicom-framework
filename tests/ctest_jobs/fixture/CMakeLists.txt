cmake_minimum_required(VERSION 3.21)
project(UmicomNotesTestFixture NONE)
enable_testing()
if(NOT EXISTS "${HELPER}")
    message(FATAL_ERROR "The compiled CTest job fixture helper is required")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/document work")
add_test(NAME notes.pass COMMAND "${HELPER}" pass)
add_test(NAME notes.fail COMMAND "${HELPER}" fail)
add_test(NAME notes.skip COMMAND "${HELPER}" skip)
set_tests_properties(notes.skip PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME "notes.save[1]+" COMMAND "${HELPER}" literal)
add_test(NAME notes.root COMMAND "${HELPER}" "root=${ROOT_LABEL}")
add_test(NAME notes.config COMMAND "${HELPER}" pass CONFIGURATIONS Debug)
add_test(NAME notes.wait COMMAND "${HELPER}" wait "${CMAKE_CURRENT_BINARY_DIR}/ready")
add_test(NAME notes.after COMMAND "${HELPER}" marker "${CMAKE_CURRENT_BINARY_DIR}/after")
add_test(NAME notes.timeout COMMAND "${HELPER}" wait "${CMAKE_CURRENT_BINARY_DIR}/ready")
set_tests_properties(notes.timeout PROPERTIES TIMEOUT 0.1)
add_test(NAME notes.large COMMAND "${HELPER}" large)
