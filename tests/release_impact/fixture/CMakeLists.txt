cmake_minimum_required(VERSION 3.24)
project(UmicomImpactMetadata NONE)
enable_testing()
include("${ROOT}/cmake/UmicomImpactInventory.cmake")
add_library(base INTERFACE)
add_library(Umicom::base ALIAS base)
add_library(app INTERFACE)
target_link_libraries(app INTERFACE Umicom::base)
if(CASE STREQUAL "imported-interface")
    add_library(external INTERFACE IMPORTED)
    set_property(TARGET external PROPERTY INTERFACE_LINK_LIBRARIES base)
    target_link_libraries(app INTERFACE external)
elseif(CASE STREQUAL "expression")
    target_link_libraries(app INTERFACE "$<$<CONFIG:Debug>:base>")
elseif(CASE STREQUAL "link-only")
    target_link_libraries(app INTERFACE "$<LINK_ONLY:base>")
endif()
_umicom_impact_target_detail(app "retained raw text" _detail)
if(NOT _detail MATCHES "dependency=62617365")
    message(FATAL_ERROR "Alias or direct dependency was omitted")
endif()
if(CASE STREQUAL "expression" OR CASE STREQUAL "imported-interface")
    if(NOT _detail MATCHES "complete=0")
        message(FATAL_ERROR "Unresolved expression was claimed complete")
    endif()
elseif(NOT _detail MATCHES "complete=1")
    message(FATAL_ERROR "Simple resolved dependency was not complete")
endif()
if(NOT _detail MATCHES "impact-end\nretained raw text$")
    message(FATAL_ERROR "Raw inventory detail was lost")
endif()
add_test(NAME sample COMMAND "${CMAKE_COMMAND}" -E true)
if(CASE STREQUAL "bad-owner")
    umicom_test_impact_owners(sample TARGETS missing)
elseif(CASE STREQUAL "bad-test")
    umicom_test_impact_owners(missing TARGETS app)
else()
    umicom_test_impact_owners(sample TARGETS app base ALWAYS)
    if(CASE STREQUAL "test-dependency")
        set_tests_properties(sample PROPERTIES DEPENDS "fixture")
    elseif(CASE STREQUAL "fixture-required")
        set_tests_properties(sample PROPERTIES FIXTURES_REQUIRED "storage")
    elseif(CASE STREQUAL "fixture-setup")
        set_tests_properties(sample PROPERTIES FIXTURES_SETUP "storage")
    elseif(CASE STREQUAL "fixture-cleanup")
        set_tests_properties(sample PROPERTIES FIXTURES_CLEANUP "storage")
    endif()
    _umicom_impact_test_detail(sample "${CMAKE_CURRENT_BINARY_DIR}" "retained test detail" _test)
    if(CMAKE_VERSION VERSION_GREATER_EQUAL "3.28")
        if(NOT _test MATCHES "target=617070" OR NOT _test MATCHES "target=62617365" OR
           NOT _test MATCHES "always=1")
            message(FATAL_ERROR "Declared test prerequisites were omitted")
        endif()
        if(CASE MATCHES "^(test-dependency|fixture-required|fixture-setup|fixture-cleanup)$")
            if(NOT _test MATCHES "complete=0")
                message(FATAL_ERROR "Unrepresented CTest dependencies were claimed complete")
            endif()
        elseif(NOT _test MATCHES "complete=1")
            message(FATAL_ERROR "Simple test ownership was not complete")
        endif()
    elseif(NOT _test STREQUAL "retained test detail")
        message(FATAL_ERROR "Older CMake must preserve registration without claiming cross-directory capture")
    endif()
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/metadata.txt" "${_detail}\n${_test}\n")
