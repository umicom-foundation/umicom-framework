cmake_minimum_required(VERSION 3.28)
project(UmicomImpactTestGraph NONE)
enable_testing()
include("${ROOT}/cmake/UmicomImpactInventory.cmake")
add_library(runner INTERFACE)
add_library(helper INTERFACE)
foreach(_test IN ITEMS alpha beta cleanup outsider)
    add_test(NAME "${_test}" COMMAND "${CMAKE_COMMAND}" -E true)
    # Give unrelated registrations distinct owners so the CLI round-trip can
    # prove that related tests were selected by graph closure, not shared code.
    #[[
    umicom_test_impact_owners("${_test}" TARGETS runner)
    ]]
    add_library("${_test}.owner" INTERFACE)
    umicom_test_impact_owners("${_test}" TARGETS "${_test}.owner")
endforeach()
if(CASE STREQUAL "dependency")
    set_tests_properties(alpha PROPERTIES DEPENDS beta)
elseif(CASE STREQUAL "fixture" OR CASE STREQUAL "multiple-setup")
    set_tests_properties(alpha PROPERTIES FIXTURES_REQUIRED storage)
    set_tests_properties(beta PROPERTIES FIXTURES_SETUP storage)
    set_tests_properties(cleanup PROPERTIES FIXTURES_CLEANUP storage)
    if(CASE STREQUAL "multiple-setup")
        set_tests_properties(outsider PROPERTIES FIXTURES_SETUP storage)
    endif()
elseif(CASE STREQUAL "cleanup-only")
    set_tests_properties(alpha PROPERTIES FIXTURES_REQUIRED storage)
    set_tests_properties(cleanup PROPERTIES FIXTURES_CLEANUP storage)
elseif(CASE STREQUAL "overlap")
    set_tests_properties(alpha PROPERTIES FIXTURES_REQUIRED first)
    set_tests_properties(beta PROPERTIES FIXTURES_SETUP first FIXTURES_REQUIRED second)
    set_tests_properties(cleanup PROPERTIES FIXTURES_CLEANUP second)
elseif(CASE STREQUAL "missing-dependency")
    set_tests_properties(alpha PROPERTIES DEPENDS absent)
elseif(CASE STREQUAL "missing-fixture")
    set_tests_properties(alpha PROPERTIES FIXTURES_REQUIRED missing)
elseif(CASE STREQUAL "expression")
    set_tests_properties(alpha PROPERTIES DEPENDS "$<IF:$<CONFIG:Debug>,beta,cleanup>")
elseif(CASE STREQUAL "self-fixture")
    set_tests_properties(alpha PROPERTIES FIXTURES_SETUP storage FIXTURES_REQUIRED storage)
elseif(CASE STREQUAL "cross-directory" OR CASE STREQUAL "duplicate-test")
    add_subdirectory(child)
    set_tests_properties(alpha PROPERTIES DEPENDS child.check)
elseif(CASE STREQUAL "false-name")
    add_test(NAME OFF COMMAND "${CMAKE_COMMAND}" -E true)
    umicom_test_impact_owners(OFF TARGETS helper)
    set_tests_properties(alpha PROPERTIES DEPENDS OFF)
elseif(CASE STREQUAL "reindex")
    umicom_impact_index_tests("${CMAKE_BINARY_DIR}")
    add_test(NAME later COMMAND "${CMAKE_COMMAND}" -E true)
    umicom_test_impact_owners(later TARGETS helper)
    set_tests_properties(alpha PROPERTIES DEPENDS later)
elseif(CASE STREQUAL "unowned")
    add_test(NAME no.owner COMMAND "${CMAKE_COMMAND}" -E true)
    set_tests_properties(alpha PROPERTIES DEPENDS no.owner)
elseif(CASE STREQUAL "fixture-scale")
    set_tests_properties(alpha PROPERTIES FIXTURES_SETUP shared)
    foreach(_i RANGE 1 1000)
        add_test(NAME "member.${_i}" COMMAND "${CMAKE_COMMAND}" -E true)
        umicom_test_impact_owners("member.${_i}" TARGETS runner)
        set_tests_properties("member.${_i}" PROPERTIES FIXTURES_REQUIRED shared)
    endforeach()
else()
    message(FATAL_ERROR "Unknown test graph fixture case: ${CASE}")
endif()
umicom_impact_index_tests("${CMAKE_BINARY_DIR}")

function(expect_row name directory complete)
    _umicom_impact_test_detail("${name}" "${directory}" "retained raw detail" _row)
    if(NOT _row MATCHES "complete=${complete}\n")
        message(FATAL_ERROR "Wrong completeness for ${name}: ${_row}")
    endif()
    if(NOT _row MATCHES "impact-end\nretained raw detail$")
        message(FATAL_ERROR "Raw detail lost for ${name}")
    endif()
    string(REGEX MATCHALL "\ntest=[0-9a-f]+\n" _actual "${_row}")
    # Check each exact edge independently; adjacent entries can share a newline.
    set(_expected "")
    foreach(_related IN LISTS ARGN)
        string(HEX "${_related}" _hex)
        if(NOT _row MATCHES "\ntest=${_hex}\n")
            message(FATAL_ERROR "Missing ${name} -> ${_related}: ${_row}")
        endif()
        list(APPEND _expected "test=${_hex}")
    endforeach()
    string(REPLACE "\n" ";" _lines "${_row}")
    set(_found "")
    foreach(_line IN LISTS _lines)
        if(_line MATCHES "^test=")
            list(APPEND _found "${_line}")
        endif()
    endforeach()
    list(SORT _expected)
    list(SORT _found)
    if(NOT _found STREQUAL _expected)
        message(FATAL_ERROR "Unexpected edges for ${name}: ${_found}; expected ${_expected}")
    endif()
    if(NOT "${_found}" STREQUAL "" AND NOT _row MATCHES "^impact-version=2\n")
        message(FATAL_ERROR "Related-test edges require compatible versioning")
    endif()
    file(APPEND "${CMAKE_BINARY_DIR}/graph-evidence.txt" "${name}\n${_row}\n")
endfunction()
file(WRITE "${CMAKE_BINARY_DIR}/graph-evidence.txt" "Case: ${CASE}\n")
if(CASE MATCHES "^(missing-dependency|missing-fixture|expression|self-fixture|duplicate-test)$")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 0)
elseif(CASE STREQUAL "dependency")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1 beta)
    expect_row(beta "${CMAKE_BINARY_DIR}" 1)
elseif(CASE STREQUAL "fixture" OR CASE STREQUAL "multiple-setup")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1)
    expect_row(beta "${CMAKE_BINARY_DIR}" 1 alpha)
    expect_row(cleanup "${CMAKE_BINARY_DIR}" 1 alpha)
    if(CASE STREQUAL "multiple-setup")
        expect_row(outsider "${CMAKE_BINARY_DIR}" 1 alpha)
    else()
        expect_row(outsider "${CMAKE_BINARY_DIR}" 1)
    endif()
elseif(CASE STREQUAL "cleanup-only")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1)
    expect_row(cleanup "${CMAKE_BINARY_DIR}" 1 alpha)
elseif(CASE STREQUAL "overlap")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1)
    expect_row(beta "${CMAKE_BINARY_DIR}" 1 alpha)
    expect_row(cleanup "${CMAKE_BINARY_DIR}" 1 beta)
elseif(CASE STREQUAL "cross-directory")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1 child.check)
    expect_row(child.check "${CMAKE_BINARY_DIR}/child" 1 cleanup)
elseif(CASE STREQUAL "false-name")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1 OFF)
elseif(CASE STREQUAL "reindex")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1 later)
    expect_row(later "${CMAKE_BINARY_DIR}" 1)
elseif(CASE STREQUAL "unowned")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1 no.owner)
    expect_row(no.owner "${CMAKE_BINARY_DIR}" 0)
elseif(CASE STREQUAL "fixture-scale")
    expect_row(alpha "${CMAKE_BINARY_DIR}" 1)
    foreach(_i RANGE 1 1000)
        expect_row("member.${_i}" "${CMAKE_BINARY_DIR}" 1 alpha)
    endforeach()
endif()

file(WRITE "${_inventory}" "UMICOM-RELEASE-INVENTORY\t1\ncontext\tcmake\t0123456789abcdef0123456789abcdef\t${_source_hex}\t${_build_hex}\t4465627567\n")
function(write_graph_rows directory)
    get_property(_targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(_target IN LISTS _targets)
        _umicom_impact_target_detail("${_target}" "Synthetic target" _detail)
        _umicom_inventory_row("${_inventory}" target "${_target}" "${CMAKE_SOURCE_DIR}" configured "${_detail}")
        _umicom_inventory_row("${_inventory}" source "${CMAKE_SOURCE_DIR}/${_target}.c" "${_target}" present "Synthetic source")
    endforeach()
    get_property(_tests DIRECTORY "${directory}" PROPERTY TESTS)
    foreach(_test IN LISTS _tests)
        _umicom_impact_test_detail("${_test}" "${directory}" "Synthetic registration" _detail)
        _umicom_inventory_row("${_inventory}" test "${_test}" "${directory}" registered "${_detail}")
    endforeach()
    get_property(_children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(_child IN LISTS _children)
        get_property(_binary DIRECTORY "${_child}" PROPERTY BINARY_DIR)
        write_graph_rows("${_binary}")
    endforeach()
endfunction()
write_graph_rows("${CMAKE_BINARY_DIR}")
