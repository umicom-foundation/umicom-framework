cmake_minimum_required(VERSION 3.24)
project(QualificationInputFixture NONE)
if(NOT DEFINED ROOT OR NOT DEFINED CASE)
    message(FATAL_ERROR "ROOT and CASE are required")
endif()
include("${ROOT}/cmake/UmicomReleaseInventory.cmake")
add_library(owner INTERFACE)
set_property(TARGET owner PROPERTY SOURCES input.h)
get_property(_original TARGET owner PROPERTY SOURCES)
set(_path "${CMAKE_CURRENT_SOURCE_DIR}/input.h")
set(_expected_rows 1)
set(_state present)
if(CASE STREQUAL "basic")
    umicom_target_qualification_inputs(owner FILES input.h)
elseif(CASE STREQUAL "duplicate")
    umicom_target_qualification_inputs(owner FILES input.h ./input.h)
    umicom_target_qualification_inputs(owner FILES "${_path}")
elseif(CASE STREQUAL "runtime")
    set(_path "${CMAKE_CURRENT_SOURCE_DIR}/runtime.ini")
    umicom_target_qualification_inputs(owner FILES runtime.ini)
elseif(CASE STREQUAL "cross-directory")
    add_subdirectory(child)
elseif(CASE STREQUAL "shared")
    add_library(second INTERFACE)
    umicom_target_qualification_inputs(owner FILES input.h)
    umicom_target_qualification_inputs(second FILES input.h)
    set(_expected_rows 2)
elseif(CASE STREQUAL "missing")
    set(_path "${CMAKE_CURRENT_BINARY_DIR}/absent.input")
    umicom_target_qualification_inputs(owner FILES "${_path}")
    set(_state unresolved)
# Retain the previous generated-source fixture for review. The replacement
# confines generated artifacts to the binary directory and uses exact identities.
#[====[
elseif(CASE STREQUAL "generated")
    set_source_files_properties(input.h PROPERTIES GENERATED TRUE)
    umicom_target_qualification_inputs(owner FILES input.h)
    set(_state unresolved)
]====]
elseif(CASE STREQUAL "generated" OR CASE STREQUAL "generated-declared-only")
    # Generated outputs belong in the binary tree; never mark a tracked header
    # GENERATED because a build-system clean may remove it. Keep this output
    # present to prove that file existence alone does not qualify generated data.
    set(_path "${CMAKE_CURRENT_BINARY_DIR}/generated.input")
    file(WRITE "${_path}" "generated fixture bytes\n")
    set_source_files_properties("${_path}" PROPERTIES GENERATED TRUE)
    if(CASE STREQUAL "generated")
        set_property(TARGET owner PROPERTY SOURCES "${_path}")
        get_property(_original TARGET owner PROPERTY SOURCES)
    endif()
    umicom_target_qualification_inputs(owner FILES "${_path}")
    set(_state unresolved)
elseif(CASE STREQUAL "bad-owner")
    umicom_target_qualification_inputs(absent FILES input.h)
elseif(CASE STREQUAL "alias")
    add_library(alias ALIAS owner)
    umicom_target_qualification_inputs(alias FILES input.h)
elseif(CASE STREQUAL "imported")
    add_library(external INTERFACE IMPORTED)
    umicom_target_qualification_inputs(external FILES input.h)
elseif(CASE STREQUAL "expression")
    umicom_target_qualification_inputs(owner FILES "$<$<BOOL:1>:input.h>")
elseif(CASE STREQUAL "directory")
    umicom_target_qualification_inputs(owner FILES "${CMAKE_CURRENT_SOURCE_DIR}")
elseif(CASE STREQUAL "no-files")
    umicom_target_qualification_inputs(owner)
elseif(CASE STREQUAL "unknown-argument")
    umicom_target_qualification_inputs(owner WRONG value FILES input.h)
else()
    message(FATAL_ERROR "Unknown fixture case")
endif()
get_property(_sources TARGET owner PROPERTY SOURCES)
if(NOT "${_sources}" STREQUAL "${_original}")
    message(FATAL_ERROR "Input declaration changed the compiler source list")
endif()
get_property(_inputs TARGET owner PROPERTY UMICOM_QUALIFICATION_INPUTS)
list(LENGTH _inputs _count)
if(NOT _count EQUAL 1)
    message(FATAL_ERROR "Declared input was lost or duplicated")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/records.tsv" "")
_umicom_inventory_directory("${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/records.tsv")
file(READ "${CMAKE_CURRENT_BINARY_DIR}/records.tsv" _records)
string(HEX "${_path}" _path_hex)
string(REGEX MATCHALL "source\t${_path_hex}\t[^\n]*" _rows "${_records}")
list(LENGTH _rows _count)
if(NOT _count EQUAL _expected_rows)
    message(FATAL_ERROR "Input capture omitted or duplicated an owned path")
endif()
string(HEX "role=declared-input\n" _role)
foreach(_row IN LISTS _rows)
    if(NOT _row MATCHES "\t${_state}\t" OR NOT _row MATCHES "${_role}")
        message(FATAL_ERROR "Input state or role was lost")
    endif()
endforeach()
if(_state STREQUAL "present")
    file(SHA256 "${_path}" _digest)
    string(HEX "sha256=${_digest}\n" _hash_field)
    foreach(_row IN LISTS _rows)
        if(NOT _row MATCHES "${_hash_field}")
            message(FATAL_ERROR "Captured input digest does not match its file")
        endif()
    endforeach()
endif()
