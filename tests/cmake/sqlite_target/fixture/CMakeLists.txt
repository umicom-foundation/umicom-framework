cmake_minimum_required(VERSION 3.24)
project(UmicomSQLiteTargetContract LANGUAGES C)
include("${RESOLVER}")
if(CASE STREQUAL "siblings")
    add_subdirectory(sibling one)
    add_subdirectory(sibling two)
    return()
elseif(CASE STREQUAL "real")
    find_package(SQLite3 REQUIRED)
elseif(CASE STREQUAL "legacy")
    add_library(SQLite::SQLite3 INTERFACE IMPORTED)
elseif(NOT CASE STREQUAL "missing")
    add_library(SQLite3::SQLite3 INTERFACE IMPORTED)
    if(CASE STREQUAL "both")
        add_library(SQLite::SQLite3 INTERFACE IMPORTED)
        set_property(TARGET SQLite::SQLite3 PROPERTY DEPRECATION "Use SQLite3::SQLite3 instead")
    endif()
endif()
umicom_resolve_sqlite_target(chosen)
if(NOT chosen STREQUAL "SQLite3::SQLite3")
    message(FATAL_ERROR "The canonical spelling was not selected")
endif()
if(CASE STREQUAL "repeated")
    umicom_resolve_sqlite_target(again)
    if(NOT chosen STREQUAL again)
        message(FATAL_ERROR "Repeated resolution changed the dependency")
    endif()
endif()
add_executable(consumer main.c)
if(CASE STREQUAL "real")
    target_compile_definitions(consumer PRIVATE UMI_REAL_SQLITE=1)
endif()
target_link_libraries(consumer PRIVATE ${chosen})
