cmake_minimum_required(VERSION 3.24)
project(FrameworkModuleComposition LANGUAGES C)
include(CTest)
set(CMAKE_INSTALL_DATADIR share)

# These imported interfaces supply names, not implementations. The fixture
# configures the real modules but deliberately does not compile or link them.
foreach(_dependency IN ITEMS developer editor platform ui_gtk4)
    add_library(Umicom::${_dependency} INTERFACE IMPORTED)
endforeach()
function(umicom_apply_warnings target)
    set_property(TARGET "${target}" PROPERTY FIXTURE_WARNINGS TRUE)
endfunction()
function(umicom_apply_sanitizers target)
    set_property(TARGET "${target}" PROPERTY FIXTURE_SANITIZERS TRUE)
endfunction()
function(umicom_register_validation_target target)
    set_property(TARGET "${target}" PROPERTY FIXTURE_VALIDATION TRUE)
endfunction()
function(require_policies target)
    foreach(_policy IN ITEMS WARNINGS SANITIZERS VALIDATION)
        get_target_property(_present "${target}" "FIXTURE_${_policy}")
        if(NOT _present)
            message(FATAL_ERROR "${target} lost the ${_policy} policy")
        endif()
    endforeach()
endfunction()

if(FIXTURE_KIND STREQUAL "diagnostic")
    option(UMICOM_BUILD_NATIVE_TOOL "Make the native tool target visible" OFF)
    if(UMICOM_BUILD_NATIVE_TOOL)
        # Only the target identity is needed by this configuration consumer.
        add_executable(umicom IMPORTED)
    endif()
    # CMakeLists.txt owns the core name. Either include order must retain both
    # distinct executables, including the OS setting with the native tool off.
    if(BUILD_TESTING AND FIXTURE_ORDER STREQUAL "core-first")
        add_executable(umicom-diagnostic-query-test "${ROOT}/tests/test_diagnostic_query.c")
    endif()
    include("${ROOT}/cmake/UmicomDiagnosticQuery.cmake")
    include("${ROOT}/cmake/UmicomDiagnosticQuery.cmake")
    if(BUILD_TESTING AND FIXTURE_ORDER STREQUAL "language-first")
        add_executable(umicom-diagnostic-query-test "${ROOT}/tests/test_diagnostic_query.c")
    endif()
    if(BUILD_TESTING)
        foreach(_target IN ITEMS umicom-diagnostic-query-test umicom-language-runtime-diagnostic-query-test)
            if(NOT TARGET "${_target}")
                message(FATAL_ERROR "Missing diagnostic owner: ${_target}")
            endif()
        endforeach()
        get_target_property(_core umicom-diagnostic-query-test SOURCES)
        get_target_property(_language umicom-language-runtime-diagnostic-query-test SOURCES)
        list(LENGTH _core _core_count)
        list(LENGTH _language _language_count)
        if(NOT _core_count EQUAL 1 OR NOT _language_count EQUAL 1
            OR NOT _core MATCHES "/tests/test_diagnostic_query.c$"
            OR NOT _language MATCHES "/tests/language_runtime/test_diagnostic_query.c$")
            message(FATAL_ERROR "Diagnostic targets no longer own their separate source files")
        endif()
        require_policies(umicom-language-runtime-diagnostic-query-test)
        foreach(_case IN ITEMS valid object encoding sync fragmented noise wrong-uri stale-version
            unversioned empty invalid-content outside surrogate related shutdown-error close-error
            timeout publication-timeout cancel-before cancel-during invalid-source state native-invalid
            read-error write-error escaped-limit server-request unknown-tag future-version)
            if(NOT TEST framework.language_runtime.diagnostic_query.${_case})
                message(FATAL_ERROR "Missing existing language diagnostic case: ${_case}")
            endif()
        endforeach()
        get_target_property(_links umicom-language-runtime-diagnostic-query-test LINK_LIBRARIES)
        foreach(_dependency IN ITEMS Umicom::developer Umicom::editor Umicom::platform)
            if(NOT _dependency IN_LIST _links)
                message(FATAL_ERROR "Missing diagnostic dependency: ${_dependency}")
            endif()
        endforeach()
    elseif(TARGET umicom-diagnostic-query-test OR TARGET umicom-language-runtime-diagnostic-query-test)
        message(FATAL_ERROR "Diagnostic tests leaked into a non-testing consumer")
    endif()
elseif(FIXTURE_KIND STREQUAL "quick-open")
    # Replace only package discovery, so these configuration checks do not need
    # GTK installed. FindPkgConfig below records the real module's request.
    list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/packages")
    if(FIXTURE_GLIB STREQUAL "existing")
        add_library(PkgConfig::GLIB INTERFACE IMPORTED)
    endif()
    if(FIXTURE_GTK)
        add_library(umicom_ui_gtk4 STATIC "${ROOT}/adapters/gtk4/quick_open_gtk4.c")
        # Clear the initial source: the attachment is responsible for adding it.
        set_property(TARGET umicom_ui_gtk4 PROPERTY SOURCES "")
    endif()
    include("${ROOT}/cmake/UmicomQuickOpen.cmake")
    umicom_attach_quick_open()
    if(FIXTURE_GTK)
        get_target_property(_sources umicom_ui_gtk4 SOURCES)
        list(LENGTH _sources _source_count)
        if(NOT _source_count EQUAL 1 OR NOT _sources MATCHES "/quick_open_gtk4.c$")
            message(FATAL_ERROR "Repeated attachment changed Quick Open source ownership")
        endif()
    endif()
    get_property(_discoveries GLOBAL PROPERTY FIXTURE_GLIB_DISCOVERIES)
    if(FIXTURE_GTK AND BUILD_TESTING)
        if(NOT TARGET PkgConfig::GLIB OR NOT TARGET umicom-quick-open-selection-test)
            message(FATAL_ERROR "Quick Open lacks its direct GLib dependency")
        endif()
        require_policies(umicom-quick-open-selection-test)
        require_policies(umicom-quick-open-native-test)
        get_target_property(_links umicom-quick-open-selection-test LINK_LIBRARIES)
        if(NOT "PkgConfig::GLIB" IN_LIST _links)
            message(FATAL_ERROR "Quick Open selection lost its GLib link")
        endif()
        if(FIXTURE_GLIB STREQUAL "discovered" AND NOT _discoveries STREQUAL "1")
            message(FATAL_ERROR "Quick Open must discover its missing GLib target exactly once")
        elseif(FIXTURE_GLIB STREQUAL "existing" AND _discoveries)
            message(FATAL_ERROR "Quick Open rediscovered a dependency supplied by its parent")
        endif()
    elseif(TARGET umicom-quick-open-selection-test OR TARGET umicom-quick-open-native-test OR _discoveries)
        message(FATAL_ERROR "Quick Open tests or discovery leaked into an inactive consumer")
    endif()
else()
    message(FATAL_ERROR "Unknown composition fixture: ${FIXTURE_KIND}")
endif()

if(NOT BUILD_TESTING)
    get_property(_tests DIRECTORY PROPERTY TESTS)
    if(_tests)
        message(FATAL_ERROR "Testing is off, but the module registered tests")
    endif()
endif()
