function(dw_case group name)
    add_test(NAME framework.desktop_workspace.${group}.${name} COMMAND umicom-desktop-workspace-test ${group} ${name})
    set_tests_properties(framework.desktop_workspace.${group}.${name} PROPERTIES TIMEOUT 60 LABELS "framework;desktop;workspace;regression")
    if(group STREQUAL "sqlite" OR group STREQUAL "local")
        set_tests_properties(framework.desktop_workspace.${group}.${name} PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endfunction()
foreach(_case IN ITEMS defaults put remove utf8 invalid_utf8 controls invalid_ids limits settings invalid_settings invalid_selection duplicate_ids alias_edit)
    dw_case(model ${_case})
endforeach()
foreach(_case IN ITEMS draft_isolation save stale revision_mismatch restore history max_snapshot clean_close unfinished external_head corrupt_chunk corrupt_digest missing_chunk capacity_rollback active_transaction)
    dw_case(storage ${_case})
endforeach()
foreach(_case IN ITEMS roundtrip truncation trailing host_layout fuzz)
    dw_case(codec ${_case})
endforeach()
foreach(_case IN ITEMS head_overflow head_sign head_schema head_zero head_tail manifest_overflow unrelated_store)
    dw_case(bad ${_case})
endforeach()
foreach(_case IN ITEMS restart write_failure rollback_failure retire_atomicity)
    dw_case(sqlite ${_case})
endforeach()
foreach(_case IN ITEMS lock reopen)
    dw_case(local ${_case})
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    foreach(_case IN ITEMS symlink private_directory database_link child_exit child_lock)
        dw_case(local ${_case})
    endforeach()
endif()

# a selected user's data to reset a test. File fixtures are ordinary directories.

    arguments "${CMAKE_CURRENT_BINARY_DIR}/arguments-${_workspace_test_id}")

    workflow "${CMAKE_CURRENT_BINARY_DIR}/workflow-${_workspace_test_id}")
