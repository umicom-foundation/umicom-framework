cmake_minimum_required(VERSION 3.24)
project(UmicomResearchReplayLaboratory VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(UMICOM_RESEARCH_SANITIZERS "Instrument the focused C library and tests" OFF)
option(UMICOM_RESEARCH_WARNINGS_AS_ERRORS "Reject focused native warnings" ON)
set(_framework "${CMAKE_CURRENT_LIST_DIR}/../..")
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(UMICOM_RESEARCH_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_RESEARCH_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
endfunction()
# Canonical source subset, not mock trading services or the complete Framework.
add_library(umicom_base STATIC "${_framework}/src/base/status.c")
add_library(umicom_trading STATIC "${_framework}/src/trading/quote.c")
add_library(umicom_trading_ui STATIC "${_framework}/src/strategy_research/backtest_engine.c")
add_library(umicom_native_launcher STATIC "${_framework}/src/native_launcher/sha256.c")
foreach(_name IN ITEMS base trading trading_ui native_launcher)
    add_library(Umicom::${_name} ALIAS umicom_${_name})
    set_target_properties(umicom_${_name} PROPERTIES EXPORT_NAME ${_name})
    target_include_directories(umicom_${_name} PUBLIC
        $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_${_name})
    umicom_apply_sanitizers(umicom_${_name})
    install(TARGETS umicom_${_name} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
foreach(_name IN ITEMS trading trading_ui native_launcher)
    target_link_libraries(umicom_${_name} PUBLIC Umicom::base)
    if(NOT WIN32)
        target_link_libraries(umicom_${_name} PUBLIC m)
    endif()
endforeach()
include("${_framework}/cmake/UmicomResearchReplay.cmake")
foreach(_header IN ITEMS base/status.h finance/types.h trading/types.h trading/quote.h
    strategy_research/backtest_engine.h native_launcher/sha256.h)
    get_filename_component(_directory "${_header}" DIRECTORY)
    install(FILES "${_framework}/include/umicom/${_header}"
        DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/umicom/${_directory}")
endforeach()
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
