cmake_minimum_required(VERSION 3.24)
project(UmicomNotesBuildPractice LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(umicom_notes_practice STATIC notes.c)
target_include_directories(umicom_notes_practice PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(umicom-notes-practice main.c)
target_link_libraries(umicom-notes-practice PRIVATE umicom_notes_practice)
if(BUILD_TESTING)
    add_executable(umicom-notes-practice-test test_notes.c)
    target_link_libraries(umicom-notes-practice-test PRIVATE umicom_notes_practice)
    add_test(NAME notes.word_count COMMAND umicom-notes-practice-test)
endif()
