cmake_minimum_required(VERSION 3.24)
project(UmicomBuildReview LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_BUILD_REVIEW_SANITIZE "Use AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(UMICOM_BUILD_REVIEW_GTK4 "Build the actual GTK review and its lifecycle tests" OFF)
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_BUILD_REVIEW_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
# These are the real canonical source files. This focused SDK does not claim
# to build the complete Framework, Studio or installed application suite.
add_library(umicom_base STATIC "${_framework}/src/base/status.c")
add_library(umicom_platform STATIC "${_framework}/src/platform/threading.c")
add_library(umicom_diagnostics STATIC
    "${_framework}/src/diagnostics/diagnostic.c" "${_framework}/src/diagnostics/compiler_parser.c")
add_library(umicom_build STATIC "${_framework}/src/build/history.c"
    "${_framework}/src/build/parser.c" "${_framework}/src/build/diagnostic.c")
add_library(umicom_compiler STATIC "${_framework}/src/compiler/diagnostic.c")
foreach(_name IN ITEMS base platform diagnostics build compiler)
    add_library(Umicom::${_name} ALIAS umicom_${_name})
    set_target_properties(umicom_${_name} PROPERTIES EXPORT_NAME ${_name})
    target_include_directories(umicom_${_name} PUBLIC
        $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_${_name})
    umicom_apply_sanitizers(umicom_${_name})
    install(TARGETS umicom_${_name} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
target_link_libraries(umicom_platform PUBLIC Umicom::base Threads::Threads)
target_link_libraries(umicom_diagnostics PUBLIC Umicom::base)
target_link_libraries(umicom_build PUBLIC Umicom::base Umicom::platform Umicom::diagnostics)
# Exercise the deferred integration entry exactly as a parent composition does.
include("${_framework}/cmake/UmicomBuildReviewIntegration.cmake")
if(UMICOM_BUILD_REVIEW_GTK4)
    include("${_framework}/cmake/UmicomBuildReview.cmake")
    include("${_framework}/cmake/UmicomBuildReviewGtk4.cmake")
endif()
foreach(_directory IN ITEMS base build compiler diagnostics)
    install(DIRECTORY "${_framework}/include/umicom/${_directory}/"
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/${_directory} FILES_MATCHING PATTERN "*.h")
endforeach()
install(FILES "${_framework}/include/umicom/platform/threading.h"
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/platform)
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomBuildReview)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomBuildReviewConfig.cmake"
    "include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomBuildReviewConfig.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomBuildReview)
