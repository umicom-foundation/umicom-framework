# Umicom Framework | Sammy Hegab, Umicom Foundation | MIT
# Canonical dependency-subset SDK for the audio laboratory, not a full app build.
cmake_minimum_required(VERSION 3.24)
project(UmicomCreativeAudio LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS NO)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(_default_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(UMICOM_FRAMEWORK_SOURCE "${_default_root}" CACHE PATH "Complete Framework source")
option(UMICOM_AUDIO_SQLITE "Canonical SQLite Data Server backend" ON)
option(UMICOM_AUDIO_SANITIZERS "Address and undefined behaviour sanitizers" OFF)
function(umicom_apply_warnings target)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_AUDIO_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
endfunction()
add_library(umicom_audio_support STATIC "${UMICOM_FRAMEWORK_SOURCE}/src/base/status.c")
set_target_properties(umicom_audio_support PROPERTIES EXPORT_NAME audio_support)
add_library(umicom_data STATIC "${UMICOM_FRAMEWORK_SOURCE}/src/data/data_server.c")
add_library(Umicom::data ALIAS umicom_data)
set_target_properties(umicom_data PROPERTIES EXPORT_NAME data)
add_library(umicom_media_engine STATIC "${UMICOM_FRAMEWORK_SOURCE}/src/media/animation_track.c")
add_library(Umicom::media_engine ALIAS umicom_media_engine)
set_target_properties(umicom_media_engine PROPERTIES EXPORT_NAME media_engine)
target_link_libraries(umicom_data PUBLIC umicom_audio_support)
target_link_libraries(umicom_media_engine PUBLIC umicom_audio_support)
if(NOT WIN32)
    target_link_libraries(umicom_media_engine PUBLIC m)
endif()
if(UMICOM_AUDIO_SQLITE)
    find_package(SQLite3 REQUIRED)
    include("${UMICOM_FRAMEWORK_SOURCE}/cmake/UmicomSQLiteTarget.cmake")
    umicom_resolve_sqlite_target(_sqlite)
    target_link_libraries(umicom_data PUBLIC ${_sqlite})
    target_compile_definitions(umicom_data PUBLIC UMICOM_HAS_SQLITE=1)
endif()
foreach(target umicom_audio_support umicom_data umicom_media_engine)
    target_include_directories(${target} PUBLIC
        $<BUILD_INTERFACE:${UMICOM_FRAMEWORK_SOURCE}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(${target})
    umicom_apply_sanitizers(${target})
    install(TARGETS ${target} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
endforeach()
include("${UMICOM_FRAMEWORK_SOURCE}/cmake/UmicomCreativeWorkspace.cmake")
foreach(path base data media)
    install(DIRECTORY "${UMICOM_FRAMEWORK_SOURCE}/include/umicom/${path}"
        DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/umicom")
endforeach()
install(FILES "${UMICOM_FRAMEWORK_SOURCE}/cmake/UmicomSQLiteTarget.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
