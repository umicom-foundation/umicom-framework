cmake_minimum_required(VERSION 3.24)
project(UmicomNotesTests LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_executable(umicom-notes-tests notes_tests.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(umicom-notes-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(BUILD_TESTING)
    add_test(NAME notes.save COMMAND umicom-notes-tests save)
    add_test(NAME notes.unicode COMMAND umicom-notes-tests unicode)
    add_test(NAME notes.index COMMAND umicom-notes-tests index)
    set_tests_properties(notes.save notes.unicode PROPERTIES TIMEOUT 5 LABELS notes)
    set_tests_properties(notes.index PROPERTIES TIMEOUT 10 LABELS "notes;index")
endif()
