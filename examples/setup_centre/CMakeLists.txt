cmake_minimum_required(VERSION 3.24)
project(UmicomSetupCentre LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_SETUP_STRICT "Treat compiler warnings as errors" ON)
option(UMICOM_SETUP_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)
option(UMICOM_SETUP_CANONICAL_PROCESS "Compile the real Framework process runner in this focused build" OFF)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        if(UMICOM_SETUP_STRICT)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(UMICOM_SETUP_STRICT)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_SETUP_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(UMICOM_SETUP_CANONICAL_PROCESS)
    # Same source files and API as the full Framework. This is a dependency-
    # subset build, not a replacement platform or a process stub.
    add_library(umicom_setup_platform STATIC
        "${_framework}/src/platform/process.c" "${_framework}/src/platform/cancellation.c")
    add_library(Umicom::platform ALIAS umicom_setup_platform)
    set_target_properties(umicom_setup_platform PROPERTIES EXPORT_NAME platform)
    target_include_directories(umicom_setup_platform PUBLIC
        $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_setup_platform)
    umicom_apply_sanitizers(umicom_setup_platform)
    install(TARGETS umicom_setup_platform EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES "${_framework}/include/umicom/platform/process.h" "${_framework}/include/umicom/platform/cancellation.h"
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/platform)
endif()
include("${_framework}/cmake/UmicomSetupCentre.cmake")
install(FILES "${_framework}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
install(EXPORT UmicomFrameworkTargets FILE UmicomSetupCentreConfig.cmake
    NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomSetupCentre)
