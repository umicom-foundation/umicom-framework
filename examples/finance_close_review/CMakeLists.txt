cmake_minimum_required(VERSION 3.24)
project(UmicomFinanceCloseReview LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_CLOSE_REVIEW_SQLITE "Use the canonical SQLite Data Server adapter" ON)
option(UMICOM_CLOSE_REVIEW_SANITIZE "Instrument this native dependency subset" OFF)
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_CLOSE_REVIEW_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
add_library(umicom_base STATIC "${_framework}/src/base/status.c")
add_library(umicom_finance STATIC "${_framework}/src/finance/identifier.c"
    "${_framework}/src/finance/core/types.c" "${_framework}/src/finance/accounting/types.c"
    "${_framework}/src/finance/accounting/journal_line.c" "${_framework}/src/finance/accounting/journal_entry.c")
add_library(umicom_data STATIC "${_framework}/src/data/data_server.c")
add_library(umicom_trading STATIC "${_framework}/src/trading/core/matching_policy.c"
    "${_framework}/src/trading/core/matching_engine.c" "${_framework}/src/broker_connectivity/ibkr_adapter.c")
foreach(_owner IN ITEMS base finance data trading)
    add_library(Umicom::${_owner} ALIAS umicom_${_owner})
    set_target_properties(umicom_${_owner} PROPERTIES EXPORT_NAME ${_owner})
    target_include_directories(umicom_${_owner} PUBLIC $<BUILD_INTERFACE:${_framework}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_${_owner})
    umicom_apply_sanitizers(umicom_${_owner})
    install(TARGETS umicom_${_owner} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
target_link_libraries(umicom_trading PUBLIC Umicom::finance Umicom::base)
set(_config "include(CMakeFindDependencyMacro)\n")
if(UMICOM_CLOSE_REVIEW_SQLITE)
    find_package(SQLite3 REQUIRED)
    include("${_framework}/cmake/UmicomSQLiteTarget.cmake")
    umicom_resolve_sqlite_target(_sqlite_target)
    target_link_libraries(umicom_data PUBLIC ${_sqlite_target})
    target_compile_definitions(umicom_data PRIVATE UMICOM_HAS_SQLITE=1)
    string(APPEND _config "find_dependency(SQLite3)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomSQLiteTarget.cmake\")\numicom_resolve_sqlite_target(_sqlite_target)\n")
endif()
include("${_framework}/cmake/UmicomFinanceOperations.cmake")
include("${_framework}/cmake/UmicomIbkrConnection.cmake")
# Only headers needed by the canonical subset are installed. No full-SDK claim.
foreach(_dir IN ITEMS base data finance trading broker_connectivity)
    install(DIRECTORY "${_framework}/include/umicom/${_dir}" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom)
endforeach()
install(FILES "${_framework}/cmake/UmicomSQLiteTarget.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFinanceCloseReview)
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFinanceCloseReview)
string(APPEND _config "include(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkTargets.cmake\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomFinanceCloseReviewConfig.cmake" "${_config}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomFinanceCloseReviewConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFinanceCloseReview)
