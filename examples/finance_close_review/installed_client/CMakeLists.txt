cmake_minimum_required(VERSION 3.24)
project(UmicomCloseReviewConsumer LANGUAGES C)
include(CTest)
find_package(UmicomFinanceCloseReview CONFIG REQUIRED)
add_executable(umicom-close-consumer main.c)
set_target_properties(umicom-close-consumer PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(umicom-close-consumer PRIVATE Umicom::finance_operations)
if(MSVC)
    target_compile_options(umicom-close-consumer PRIVATE /W4 /WX)
else()
    target_compile_options(umicom-close-consumer PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME finance_close_review.installed_consumer COMMAND umicom-close-consumer)
