cmake_minimum_required(VERSION 3.28)
project(UmicomFormRecordConsumer LANGUAGES C)
# The example resolves the installed SDK only; no Framework source fallback.
find_package(UmicomFramework CONFIG REQUIRED COMPONENTS application data)
if(NOT TARGET Umicom::form_records)
    message(FATAL_ERROR "This Framework installation does not include the form_records adapter")
endif()
add_executable(umicom-form-record-example main.c)
target_link_libraries(umicom-form-record-example PRIVATE Umicom::form_records)
set_target_properties(umicom-form-record-example PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME consumer.form-record-review COMMAND umicom-form-record-example)
endif()
if(TARGET Umicom::form_record_editor_gtk4)
    add_executable(umicom-form-record-native native.c)
    target_link_libraries(umicom-form-record-native PRIVATE Umicom::form_record_editor_gtk4)
    set_target_properties(umicom-form-record-native PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
