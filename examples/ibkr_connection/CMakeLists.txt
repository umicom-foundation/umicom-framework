cmake_minimum_required(VERSION 3.24)
project(UmicomBrokerConnectionLaboratory VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(_framework "${CMAKE_CURRENT_LIST_DIR}/../..")
option(UMICOM_IBKR_GTK4 "Build the native GTK connection workbench when available" ON)
option(UMICOM_IBKR_SANITIZERS "Enable address and undefined-behaviour sanitizers" OFF)
option(UMICOM_IBKR_WARNINGS_AS_ERRORS "Treat focused compilation warnings as errors" ON)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(UMICOM_IBKR_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_IBKR_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
endfunction()
# Actual status and broker-mapping source. This small SDK is a dependency
# subset for testing, not a replacement definition of the full trading platform.
add_library(umicom_base STATIC "${_framework}/src/base/status.c")
add_library(umicom_trading STATIC "${_framework}/src/broker_connectivity/ibkr_adapter.c")
foreach(_name IN ITEMS base trading)
    add_library(Umicom::${_name} ALIAS umicom_${_name})
    set_target_properties(umicom_${_name} PROPERTIES EXPORT_NAME ${_name})
    target_include_directories(umicom_${_name} PUBLIC
        $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_${_name})
    umicom_apply_sanitizers(umicom_${_name})
    install(TARGETS umicom_${_name} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
target_link_libraries(umicom_trading PUBLIC Umicom::base)
include("${_framework}/cmake/UmicomIbkrConnection.cmake")
if(UMICOM_IBKR_GTK4)
    include("${_framework}/cmake/UmicomIbkrConnectionGtk4.cmake")
endif()
foreach(_header IN ITEMS base/status.h finance/types.h trading/types.h broker_connectivity/ibkr_adapter.h)
    get_filename_component(_dir "${_header}" DIRECTORY)
    install(FILES "${_framework}/include/umicom/${_header}" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/umicom/${_dir}")
endforeach()
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
