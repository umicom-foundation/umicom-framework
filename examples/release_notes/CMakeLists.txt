# A small real Windows DLL/application pair for installation acceptance.
if(NOT WIN32)
    message(STATUS "Windows Notes release laboratory not built on this host")
    return()
endif()
enable_language(RC)
add_library(umicom-release-notes-model SHARED model.c)
target_compile_definitions(umicom-release-notes-model PRIVATE UMICOM_RELEASE_NOTES_EXPORTS=1)
set_target_properties(umicom-release-notes-model PROPERTIES PREFIX "" C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_executable(umicom-release-notes WIN32 main_windows.c notes.rc)
target_link_libraries(umicom-release-notes PRIVATE umicom-release-notes-model user32 gdi32 comdlg32)
target_compile_definitions(umicom-release-notes PRIVATE UNICODE _UNICODE)
target_include_directories(umicom-release-notes PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../setup_centre/resources")
set_target_properties(umicom-release-notes PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MINGW)
    target_link_options(umicom-release-notes PRIVATE -municode -static-libgcc)
    target_link_options(umicom-release-notes-model PRIVATE -static-libgcc)
endif()
foreach(_target IN ITEMS umicom-release-notes umicom-release-notes-model)
    if(COMMAND umicom_apply_warnings)
        umicom_apply_warnings(${_target})
    endif()
    if(COMMAND umicom_apply_sanitizers)
        umicom_apply_sanitizers(${_target})
    endif()
endforeach()
add_custom_command(TARGET umicom-release-notes POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/example-note.txt"
        "$<TARGET_FILE_DIR:umicom-release-notes>/example-note.txt" VERBATIM)
