cmake_minimum_required(VERSION 3.24)
project(UmicomDataSafety LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_DATA_SAFETY_SQLITE "Compile the actual SQLite backend" ON)
option(UMICOM_DATA_SAFETY_SANITIZE "Enable address and undefined behaviour checks" OFF)
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_DATA_SAFETY_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
# Dependency-subset build: the canonical Data Server source, not a replacement.
add_library(umicom_data STATIC "${_framework}/src/data/data_server.c")
add_library(Umicom::data ALIAS umicom_data)
set_target_properties(umicom_data PROPERTIES EXPORT_NAME data)
target_include_directories(umicom_data PUBLIC $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(UMICOM_DATA_SAFETY_SQLITE)
    find_package(SQLite3 QUIET)
    if(SQLite3_FOUND)
        target_compile_definitions(umicom_data PRIVATE UMICOM_HAS_SQLITE=1)
        target_link_libraries(umicom_data PUBLIC SQLite::SQLite3)
    endif()
endif()
umicom_apply_warnings(umicom_data)
umicom_apply_sanitizers(umicom_data)
include("${_framework}/cmake/UmicomDataSafety.cmake")
install(TARGETS umicom_data EXPORT UmicomDataSafetyTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES "${_framework}/include/umicom/data/data_server.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/data)
install(FILES "${_framework}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
install(EXPORT UmicomDataSafetyTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomDataSafety)
set(_config "include(CMakeFindDependencyMacro)\n")
if(UMICOM_DATA_SAFETY_SQLITE AND SQLite3_FOUND)
    string(APPEND _config "find_dependency(SQLite3)\n")
endif()
string(APPEND _config "include(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomDataSafetyTargets.cmake\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomDataSafetyConfig.cmake" "${_config}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomDataSafetyConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomDataSafety)
