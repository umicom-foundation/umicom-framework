# Standalone native build: the complete delivery library, not a mock runtime.
cmake_minimum_required(VERSION 3.24)
project(UmicomNativeLauncher LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
include(CTest)
include(GNUInstallDirs)
option(UMICOM_NATIVE_LAUNCHER_STRICT "Treat warnings as errors" ON)
option(UMICOM_NATIVE_LAUNCHER_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        if(UMICOM_NATIVE_LAUNCHER_STRICT)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(UMICOM_NATIVE_LAUNCHER_STRICT)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_NATIVE_LAUNCHER_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${_root}/cmake/UmicomNativeLauncher.cmake")
# The optional include is only for replaying a source-only delivery outside a
# complete checkout. It must contain the real, recorded Framework status.h.
set(UMICOM_NATIVE_LAUNCHER_CONTRACT_INCLUDE "" CACHE PATH "Canonical Framework include directory for a delivery replay")
if(UMICOM_NATIVE_LAUNCHER_CONTRACT_INCLUDE)
    target_include_directories(umicom_native_launcher PUBLIC
        $<BUILD_INTERFACE:${UMICOM_NATIVE_LAUNCHER_CONTRACT_INCLUDE}>)
    install(FILES "${UMICOM_NATIVE_LAUNCHER_CONTRACT_INCLUDE}/umicom/base/status.h"
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
else()
    install(FILES "${_root}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
endif()
install(EXPORT UmicomFrameworkTargets FILE UmicomNativeLauncherConfig.cmake
    NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomNativeLauncher)
