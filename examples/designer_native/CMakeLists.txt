cmake_minimum_required(VERSION 3.24)
project(UmicomDesignerNative LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_DESIGNER_NATIVE_SANITIZE "Enable address/undefined sanitizers" OFF)
option(UMICOM_DESIGNER_NATIVE_GTK4 "Build the real GTK adapter and graphical tests" OFF)
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_DESIGNER_NATIVE_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
endfunction()
# Compile canonical production sources, not substitute document implementations.
# This SDK is a dependency subset, not a full Framework or Studio build.
add_library(umicom_base STATIC "${_framework}/src/base/status.c")
add_library(umicom_declarative STATIC
    "${_framework}/src/declarative/types.c" "${_framework}/src/declarative/value.c"
    "${_framework}/src/declarative/attribute.c" "${_framework}/src/declarative/node.c"
    "${_framework}/src/declarative/document.c" "${_framework}/src/declarative/lexer.c"
    "${_framework}/src/declarative/parser.c" "${_framework}/src/declarative/diagnostic.c")
foreach(_name IN ITEMS base declarative)
    add_library(Umicom::${_name} ALIAS umicom_${_name})
    set_target_properties(umicom_${_name} PROPERTIES EXPORT_NAME ${_name})
    target_include_directories(umicom_${_name} PUBLIC
        $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_${_name})
    umicom_apply_sanitizers(umicom_${_name})
    install(TARGETS umicom_${_name} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
target_link_libraries(umicom_declarative PUBLIC Umicom::base)
# Exercise the same delayed composition as the complete repository.
include("${_framework}/cmake/UmicomDesignerNativeIntegration.cmake")
if(UMICOM_DESIGNER_NATIVE_GTK4)
    include("${_framework}/cmake/UmicomDesignerNativeGtk4.cmake")
endif()
install(FILES "${_framework}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
foreach(_name IN ITEMS types value attribute node document lexer parser diagnostic)
    install(FILES "${_framework}/include/umicom/declarative/${_name}.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/declarative)
endforeach()
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomDesignerNative)
set(_dependencies "")
if(UMICOM_DESIGNER_NATIVE_GTK4)
    set(_dependencies "include(CMakeFindDependencyMacro)\nfind_dependency(PkgConfig)\npkg_check_modules(UMICOM_DESIGNER_NATIVE_GTK REQUIRED IMPORTED_TARGET GLOBAL gtk4>=4.10)\n")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomDesignerNativeConfig.cmake"
    "${_dependencies}include(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomDesignerNativeConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomDesignerNative)
