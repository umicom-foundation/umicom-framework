cmake_minimum_required(VERSION 3.28)
project(UmicomBrowserFormExample LANGUAGES C)
find_package(UmicomFramework CONFIG REQUIRED COMPONENTS application web document)
if(NOT TARGET Umicom::form_web)
    message(FATAL_ERROR "This Framework installation does not include the form_web adapter")
endif()
add_executable(umicom-browser-form main.c)
target_link_libraries(umicom-browser-form PRIVATE Umicom::form_web)
set_target_properties(umicom-browser-form PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
enable_testing()
add_test(NAME browser_form_review COMMAND umicom-browser-form)
