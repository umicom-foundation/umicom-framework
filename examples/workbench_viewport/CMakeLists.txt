cmake_minimum_required(VERSION 3.24)
project(UmicomWorkbenchViewport LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_VIEWPORT_SANITIZE "Address and undefined-behaviour checks" OFF)
option(UMICOM_VIEWPORT_GTK4 "Build the actual GTK4 component gallery" OFF)
option(UMICOM_VIEWPORT_DATA_TESTS "Exercise the existing canonical Data Server tests" ON)
option(UMICOM_DATA_SAFETY_SQLITE "Include the real SQLite backend when found" ON)
get_filename_component(_framework "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_VIEWPORT_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
include("${_framework}/cmake/UmicomWorkbenchViewport.cmake")
# The focused build uses the actual canonical Data Server and existing tests.
# It does not manufacture a complete Framework platform or an application host.
if(UMICOM_VIEWPORT_DATA_TESTS)
    add_library(umicom_data STATIC "${_framework}/src/data/data_server.c")
    add_library(Umicom::data ALIAS umicom_data)
    target_include_directories(umicom_data PUBLIC "${_framework}/include")
    if(UMICOM_DATA_SAFETY_SQLITE)
        find_package(SQLite3 QUIET)
        if(SQLite3_FOUND)
            include("${_framework}/cmake/UmicomSQLiteTarget.cmake")
            umicom_resolve_sqlite_target(_sqlite)
            target_compile_definitions(umicom_data PRIVATE UMICOM_HAS_SQLITE=1)
            target_link_libraries(umicom_data PUBLIC ${_sqlite})
        endif()
    endif()
    umicom_apply_warnings(umicom_data)
    umicom_apply_sanitizers(umicom_data)
    include("${_framework}/cmake/UmicomDataSafety.cmake")
endif()
if(UMICOM_VIEWPORT_GTK4)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(VIEWPORT_GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
    include("${_framework}/cmake/UmicomWorkbenchViewportGtk4.cmake")
endif()
# Small, honest SDK for the new header-based projection target only.
install(FILES "${_framework}/include/umicom/workbench_layout/types.h"
    "${_framework}/include/umicom/workbench_layout/node.h"
    "${_framework}/include/umicom/workbench_layout/document.h"
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/workbench_layout)
install(FILES "${_framework}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomViewport)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomViewportConfig.cmake"
    "include(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomViewportConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomViewport)
