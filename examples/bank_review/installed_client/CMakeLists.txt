cmake_minimum_required(VERSION 3.24)
project(UmicomInstalledBankReview C)
include(CTest)
find_package(UmicomBankReview CONFIG REQUIRED)
add_executable(umicom-installed-bank-review main.c)
set_target_properties(umicom-installed-bank-review PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_link_libraries(umicom-installed-bank-review PRIVATE Umicom::bank_operations)
if(MSVC)
    target_compile_options(umicom-installed-bank-review PRIVATE /W4 /WX)
else()
    target_compile_options(umicom-installed-bank-review PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME installed.bank_review COMMAND umicom-installed-bank-review)
