cmake_minimum_required(VERSION 3.24)
project(UmicomMarketTapeLaboratory VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
set(_framework "${CMAKE_CURRENT_LIST_DIR}/../..")
option(UMICOM_MARKET_TAPE_GTK4 "Build the GTK market-tape practice workspace when available" ON)
option(UMICOM_MARKET_TAPE_SANITIZERS "Instrument focused native targets" OFF)
option(UMICOM_MARKET_TAPE_WARNINGS_AS_ERRORS "Treat focused native warnings as errors" ON)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(UMICOM_MARKET_TAPE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_MARKET_TAPE_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
endfunction()
# Real canonical dependencies only. This SDK is deliberately a subset, not a
# substitute for the full Framework/Trader build and installed-product checks.
add_library(umicom_base STATIC "${_framework}/src/base/status.c")
add_library(umicom_platform STATIC "${_framework}/src/platform/threading.c")
add_library(umicom_trading STATIC "${_framework}/src/trading/quote.c"
    "${_framework}/src/trading/market_data_quality.c")
foreach(_name IN ITEMS base platform trading)
    add_library(Umicom::${_name} ALIAS umicom_${_name})
    set_target_properties(umicom_${_name} PROPERTIES EXPORT_NAME ${_name})
    target_include_directories(umicom_${_name} PUBLIC
        $<BUILD_INTERFACE:${_framework}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_${_name})
    umicom_apply_sanitizers(umicom_${_name})
    install(TARGETS umicom_${_name} EXPORT UmicomFrameworkTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
target_link_libraries(umicom_platform PUBLIC Umicom::base Threads::Threads)
target_link_libraries(umicom_trading PUBLIC Umicom::base)
if(NOT WIN32)
    target_link_libraries(umicom_trading PUBLIC m)
endif()
include("${_framework}/cmake/UmicomMarketTape.cmake")
if(UMICOM_MARKET_TAPE_GTK4)
    include("${_framework}/cmake/UmicomMarketTapeGtk4.cmake")
endif()
foreach(_header IN ITEMS base/status.h platform/threading.h finance/types.h trading/types.h
        trading/quote.h trading/market_data_quality.h)
    get_filename_component(_dir "${_header}" DIRECTORY)
    install(FILES "${_framework}/include/umicom/${_header}" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/umicom/${_dir}")
endforeach()
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework)
