cmake_minimum_required(VERSION 3.25)
project(UmicomBootMediaTools LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
get_filename_component(UMICOM_FRAMEWORK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_BOOT_MEDIA_SANITIZE "Address and undefined-behaviour sanitizers" OFF)
function(umicom_apply_warnings target)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_BOOT_MEDIA_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
# This focused platform SDK uses actual canonical process and boot-report
# sources; it is not the complete Framework platform or Applications build.
add_library(umicom_media_platform STATIC "${UMICOM_FRAMEWORK_ROOT}/src/platform/process.c"
    "${UMICOM_FRAMEWORK_ROOT}/src/platform/cancellation.c" "${UMICOM_FRAMEWORK_ROOT}/src/platform/boot_report.c")
add_library(Umicom::platform ALIAS umicom_media_platform)
set_target_properties(umicom_media_platform PROPERTIES EXPORT_NAME platform)
target_include_directories(umicom_media_platform PUBLIC $<BUILD_INTERFACE:${UMICOM_FRAMEWORK_ROOT}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(umicom_media_platform PUBLIC Threads::Threads)
umicom_apply_warnings(umicom_media_platform)
umicom_apply_sanitizers(umicom_media_platform)
install(TARGETS umicom_media_platform EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES "${UMICOM_FRAMEWORK_ROOT}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
install(FILES "${UMICOM_FRAMEWORK_ROOT}/include/umicom/platform/process.h" "${UMICOM_FRAMEWORK_ROOT}/include/umicom/platform/cancellation.h"
    "${UMICOM_FRAMEWORK_ROOT}/include/umicom/platform/boot_report.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/platform)
include("${UMICOM_FRAMEWORK_ROOT}/cmake/UmicomBootMedia.cmake")
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomMediaSDK)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomMediaSDKConfig.cmake" "include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomMediaSDKConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomMediaSDK)
