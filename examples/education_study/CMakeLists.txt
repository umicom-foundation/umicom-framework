cmake_minimum_required(VERSION 3.25)
project(UmicomEducationStudy LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
get_filename_component(FW "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_STUDY_SQLITE "Enable the real Data Server SQLite backend" ON)
option(UMICOM_STUDY_SANITIZERS "Instrument the focused native targets" OFF)
function(umicom_apply_warnings target)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_STUDY_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
add_library(umicom_data STATIC "${FW}/src/data/data_server.c" "${FW}/src/base/status.c")
add_library(Umicom::data ALIAS umicom_data)
set_target_properties(umicom_data PROPERTIES EXPORT_NAME data C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(umicom_data PUBLIC $<BUILD_INTERFACE:${FW}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(UMICOM_STUDY_SQLITE)
    find_package(SQLite3 REQUIRED)
    include("${FW}/cmake/UmicomSQLiteTarget.cmake")
    umicom_resolve_sqlite_target(sqlite_target)
    target_compile_definitions(umicom_data PUBLIC UMICOM_HAS_SQLITE=1)
    target_link_libraries(umicom_data PUBLIC ${sqlite_target})
endif()
add_library(umicom_developer STATIC "${FW}/src/teacher/types.c" "${FW}/src/teacher/assessment_scoring.c")
add_library(Umicom::developer ALIAS umicom_developer)
set_target_properties(umicom_developer PROPERTIES EXPORT_NAME developer C_STANDARD 23 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(umicom_developer PUBLIC $<BUILD_INTERFACE:${FW}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
foreach(target umicom_data umicom_developer)
    umicom_apply_warnings(${target})
    umicom_apply_sanitizers(${target})
endforeach()
include("${FW}/cmake/UmicomEducationWorkspace.cmake")
install(TARGETS umicom_data umicom_developer EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(DIRECTORY "${FW}/include/umicom/base" "${FW}/include/umicom/data" "${FW}/include/umicom/teacher"
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/umicom")
install(FILES "${FW}/cmake/UmicomSQLiteTarget.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomFrameworkConfig.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/UmicomFramework")
