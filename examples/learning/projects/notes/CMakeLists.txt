cmake_minimum_required(VERSION 3.24)
project(UmicomLesson LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
function(lesson_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
add_library(notes STATIC notes.c)
target_include_directories(notes PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
lesson_warnings(notes)
add_executable(umicom-notes main.c)
target_link_libraries(umicom-notes PRIVATE notes)
lesson_warnings(umicom-notes)
if(BUILD_TESTING)
    add_executable(notes-checks test_notes.c)
    target_link_libraries(notes-checks PRIVATE notes)
    lesson_warnings(notes-checks)
    add_test(NAME lesson.notes.contracts COMMAND notes-checks)
    set_tests_properties(lesson.notes.contracts PROPERTIES TIMEOUT 15)
endif()
