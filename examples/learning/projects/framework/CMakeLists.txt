cmake_minimum_required(VERSION 3.24)
project(UmicomLesson LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
function(lesson_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
# Supply CMAKE_PREFIX_PATH for an installed Umicom Framework SDK.
if(NOT TARGET Umicom::data)
    find_package(UmicomFramework CONFIG REQUIRED COMPONENTS data)
endif()
add_executable(umicom-framework-notes main.c)
target_link_libraries(umicom-framework-notes PRIVATE Umicom::data)
lesson_warnings(umicom-framework-notes)
if(BUILD_TESTING)
    add_executable(framework-notes-checks test_storage.c)
    target_link_libraries(framework-notes-checks PRIVATE Umicom::data)
    lesson_warnings(framework-notes-checks)
    add_test(NAME lesson.framework.storage COMMAND framework-notes-checks)
    set_tests_properties(lesson.framework.storage PROPERTIES TIMEOUT 15)
endif()
option(UMICOM_LESSON_WITH_GTK "Build the optional GTK notes view" OFF)
if(UMICOM_LESSON_WITH_GTK)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LESSON_GTK REQUIRED IMPORTED_TARGET gtk4>=4.6)
    add_executable(umicom-framework-notes-window window.c)
    target_link_libraries(umicom-framework-notes-window PRIVATE Umicom::data PkgConfig::LESSON_GTK)
    lesson_warnings(umicom-framework-notes-window)
endif()
