cmake_minimum_required(VERSION 3.24)
project(UmicomLesson LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
function(lesson_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endfunction()
option(UMICOM_LESSON_USE_ASSEMBLY "Use the supported x86-64 GNU Assembly routine" ON)
add_library(stock_total STATIC stock_total.c)
target_include_directories(stock_total PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(UMICOM_LESSON_USE_ASSEMBLY AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND
   CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$" AND
   CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$" AND NOT APPLE)
    enable_language(ASM)
    target_sources(stock_total PRIVATE stock_total.S)
    target_compile_definitions(stock_total PUBLIC UMICOM_LESSON_NATIVE_ASSEMBLY=1)
    message(STATUS "Stock total: native x86-64 Assembly selected")
else()
    message(STATUS "Stock total: C reference selected; Assembly is not tested by this build")
endif()
lesson_warnings(stock_total)
add_executable(umicom-stock-total main.c)
target_link_libraries(umicom-stock-total PRIVATE stock_total)
lesson_warnings(umicom-stock-total)
if(BUILD_TESTING)
    add_executable(stock-total-checks test_total.c)
    target_link_libraries(stock-total-checks PRIVATE stock_total)
    lesson_warnings(stock-total-checks)
    add_test(NAME lesson.assembly.comparison COMMAND stock-total-checks)
    set_tests_properties(lesson.assembly.comparison PROPERTIES TIMEOUT 15)
endif()
