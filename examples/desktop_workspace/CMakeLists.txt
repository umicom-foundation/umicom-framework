cmake_minimum_required(VERSION 3.24)
project(UmicomDesktopWorkspaceChecks LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
include(GNUInstallDirs)
get_filename_component(UMICOM_WORKSPACE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
option(UMICOM_WORKSPACE_STRICT "Treat focused-project compiler warnings as errors" ON)
option(UMICOM_WORKSPACE_SANITIZE "Run address and undefined-behaviour sanitizers" OFF)
option(UMICOM_WORKSPACE_SQLITE "Use the available canonical SQLite backend" ON)
function(umicom_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(UMICOM_WORKSPACE_STRICT)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
function(umicom_apply_sanitizers target)
    if(UMICOM_WORKSPACE_SANITIZE AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
# This focused project compiles the ACTUAL Data Server and SHA-256 sources.
# These dependency subsets are not a build of the complete Framework runtime.
if(NOT TARGET Umicom::data)
    add_library(umicom_workspace_data STATIC "${UMICOM_WORKSPACE_SOURCE}/src/data/data_server.c")
    add_library(Umicom::data ALIAS umicom_workspace_data)
    set_target_properties(umicom_workspace_data PROPERTIES EXPORT_NAME data)
    target_include_directories(umicom_workspace_data PUBLIC
        $<BUILD_INTERFACE:${UMICOM_WORKSPACE_SOURCE}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    if(UMICOM_WORKSPACE_SQLITE)
        find_package(SQLite3 QUIET)
        if(SQLite3_FOUND)
            target_compile_definitions(umicom_workspace_data PRIVATE UMICOM_HAS_SQLITE=1)
            target_link_libraries(umicom_workspace_data PUBLIC SQLite::SQLite3)
        endif()
    endif()
    umicom_apply_warnings(umicom_workspace_data)
    umicom_apply_sanitizers(umicom_workspace_data)
    install(TARGETS umicom_workspace_data EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(NOT TARGET Umicom::native_launcher)
    add_library(umicom_workspace_digest STATIC "${UMICOM_WORKSPACE_SOURCE}/src/native_launcher/sha256.c")
    add_library(Umicom::native_launcher ALIAS umicom_workspace_digest)
    set_target_properties(umicom_workspace_digest PROPERTIES EXPORT_NAME native_launcher)
    target_include_directories(umicom_workspace_digest PUBLIC
        $<BUILD_INTERFACE:${UMICOM_WORKSPACE_SOURCE}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    umicom_apply_warnings(umicom_workspace_digest)
    umicom_apply_sanitizers(umicom_workspace_digest)
    install(TARGETS umicom_workspace_digest EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
include("${UMICOM_WORKSPACE_SOURCE}/cmake/UmicomDesktopWorkspace.cmake")
install(FILES "${UMICOM_WORKSPACE_SOURCE}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
install(FILES "${UMICOM_WORKSPACE_SOURCE}/include/umicom/data/data_server.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/data)
install(FILES "${UMICOM_WORKSPACE_SOURCE}/include/umicom/native_launcher/sha256.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/native_launcher)
set(_deps "include(CMakeFindDependencyMacro)\n")
if(UMICOM_WORKSPACE_SQLITE AND SQLite3_FOUND)
    string(APPEND _deps "find_dependency(SQLite3)\n")
endif()
string(APPEND _deps "include(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomDesktopWorkspaceTargets.cmake\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomDesktopWorkspaceConfig.cmake" "${_deps}")
install(EXPORT UmicomFrameworkTargets FILE UmicomDesktopWorkspaceTargets.cmake NAMESPACE Umicom::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomDesktopWorkspace)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomDesktopWorkspaceConfig.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomDesktopWorkspace)
