cmake_minimum_required(VERSION 3.25)
project(UmicomVmManagerTools LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
get_filename_component(UMICOM_FRAMEWORK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(UMICOM_VM_SANITIZE "Address/undefined-behaviour sanitizers" OFF)
option(UMICOM_VM_SQLITE "Compile real SQLite profile persistence" ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(UMICOM_VM_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
# Focused project applies strict flags globally, including dependency tests.
function(umicom_apply_warnings target)
endfunction()
function(umicom_apply_sanitizers target)
endfunction()
# Canonical sources, with a deliberately narrow platform/Data Server SDK.
add_library(umicom_vm_platform STATIC "${UMICOM_FRAMEWORK_ROOT}/src/platform/process.c"
    "${UMICOM_FRAMEWORK_ROOT}/src/platform/cancellation.c" "${UMICOM_FRAMEWORK_ROOT}/src/platform/boot_report.c")
add_library(Umicom::platform ALIAS umicom_vm_platform)
set_target_properties(umicom_vm_platform PROPERTIES EXPORT_NAME platform)
target_link_libraries(umicom_vm_platform PUBLIC Threads::Threads)
add_library(umicom_vm_data STATIC "${UMICOM_FRAMEWORK_ROOT}/src/data/data_server.c")
add_library(Umicom::data ALIAS umicom_vm_data)
set_target_properties(umicom_vm_data PROPERTIES EXPORT_NAME data)
target_link_libraries(umicom_vm_data PUBLIC Threads::Threads)
set(_sdk_sqlite "")
if(UMICOM_VM_SQLITE)
    find_package(SQLite3 REQUIRED)
    target_compile_definitions(umicom_vm_data PRIVATE UMICOM_HAS_SQLITE=1)
    target_link_libraries(umicom_vm_data PUBLIC SQLite::SQLite3)
    set(_sdk_sqlite "find_dependency(SQLite3)\n")
endif()
foreach(_target IN ITEMS umicom_vm_platform umicom_vm_data)
    target_include_directories(${_target} PUBLIC $<BUILD_INTERFACE:${UMICOM_FRAMEWORK_ROOT}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    install(TARGETS ${_target} EXPORT UmicomFrameworkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endforeach()
install(FILES "${UMICOM_FRAMEWORK_ROOT}/include/umicom/base/status.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/base)
install(FILES "${UMICOM_FRAMEWORK_ROOT}/include/umicom/data/data_server.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/data)
install(FILES "${UMICOM_FRAMEWORK_ROOT}/include/umicom/platform/process.h" "${UMICOM_FRAMEWORK_ROOT}/include/umicom/platform/cancellation.h"
    "${UMICOM_FRAMEWORK_ROOT}/include/umicom/platform/boot_report.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/umicom/platform)
include("${UMICOM_FRAMEWORK_ROOT}/cmake/UmicomVmManager.cmake")
install(EXPORT UmicomFrameworkTargets NAMESPACE Umicom:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomVmSDK)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/UmicomVmSDKConfig.cmake"
    "include(CMakeFindDependencyMacro)\nfind_dependency(Threads)\n${_sdk_sqlite}include(\"\${CMAKE_CURRENT_LIST_DIR}/UmicomFrameworkTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/UmicomVmSDKConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UmicomVmSDK)
